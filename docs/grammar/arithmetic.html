expression

expression ::= or-expr

or-expr

or-expr ::= and-expr ( '||' and-expr )*

and-expr

and-expr ::= bitwise-or-expr ( '&&' bitwise-or-expr )*

bitwise-or-expr

bitwise-or-expr ::= bitwise-xor-expr ( '|' bitwise-xor-expr )*

bitwise-xor-expr

bitwise-xor-expr ::= bitwise-and-expr ( '^' bitwise-and-expr )*

bitwise-and-expr

bitwise-and-expr ::= equality-expr ( '&' equality-expr )*

equality-expr

equality-expr ::= relational-expr ( ( '==' | '!=' ) relational-expr )*

relational-expr

relational-expr ::= shift-expr ( ( '<=' | '>=' | '<' | '>' ) shift-expr )*

shift-expr

shift-expr ::= additive-expr ( ( '>>>' | '<<' | '>>' ) additive-expr )*

additive-expr

additive-expr ::= multiplicative-expr ( ( '+' | '-' ) multiplicative-expr )*

multiplicative-expr

multiplicative-expr ::= unary-expr ( ( '*' | '/' | '%' ) unary-expr )*

unary-expr

unary-expr ::= ( '!' | '-' | '~' | '+' ) unary-expr | primary

primary

primary ::= float-literal | hex-literal | bin-literal | oct-literal | int-literal | bool-literal | function-call | identifier | '(' expression ')'

int-literal

int-literal ::= /[0-9]+/

float-literal

float-literal ::= /[0-9]+\.[0-9]*([eE][+-]?[0-9]+)?/ | /\.[0-9]+([eE][+-]?[0-9]+)?/ | /[0-9]+[eE][+-]?[0-9]+/

bool-literal

bool-literal ::= 'true' | 'false'

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

function-call

function-call ::= identifier '(' arg-list? ')'

arg-list

arg-list ::= expression ( ',' expression )*
