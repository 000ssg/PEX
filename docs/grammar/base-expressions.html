program

program ::= statement+

statement

statement ::= assignment | conditional | loop | return-statement | expression

block

block ::= '{' statement* '}'

assignment

assignment ::= identifier '=' expression

conditional

conditional ::= 'if' '(' expression ')' block ( 'else' block )?

loop

loop ::= 'while' '(' expression ')' block

return-statement

return-statement ::= 'return' expression?

expression

expression ::= or-expr

or-expr

or-expr ::= and-expr ( '||' and-expr )*

and-expr

and-expr ::= bitwise-or-expr ( '&&' bitwise-or-expr )*

bitwise-or-expr

bitwise-or-expr ::= bitwise-xor-expr ( '|' bitwise-xor-expr )*

bitwise-xor-expr

bitwise-xor-expr ::= bitwise-and-expr ( '^' bitwise-and-expr )*

bitwise-and-expr

bitwise-and-expr ::= equality-expr ( '&' equality-expr )*

equality-expr

equality-expr ::= relational-expr ( ( '==' | '!=' ) relational-expr )*

relational-expr

relational-expr ::= shift-expr ( ( '<=' | '>=' | '<' | '>' ) shift-expr )*

shift-expr

shift-expr ::= addition-expr ( ( '>>>' | '<<' | '>>' ) addition-expr )*

addition-expr

addition-expr ::= multiplication-expr ( ( '+' | '-' ) multiplication-expr )*

multiplication-expr

multiplication-expr ::= unary-expr ( ( '*' | '/' | '%' ) unary-expr )*

unary-expr

unary-expr ::= ( '!' | '-' | '~' | '+' ) unary-expr | primary

primary

primary ::= literal | function-call | identifier | index-access | '(' expression ')'

index-access

index-access ::= identifier '[' expression ']'
