literal

literal ::= float-literal | hex-literal | bin-literal | oct-literal | int-literal | bool-literal | string-literal | null-literal

int-literal

int-literal ::= /[0-9]+/

float-literal

float-literal ::= /[0-9]+\.[0-9]*([eE][+-]?[0-9]+)?/ | /\.[0-9]+([eE][+-]?[0-9]+)?/ | /[0-9]+[eE][+-]?[0-9]+/

hex-literal

hex-literal ::= /0[xX][0-9a-fA-F]+/

bin-literal

bin-literal ::= /0[bB][01]+/

oct-literal

oct-literal ::= /0[oO][0-7]+/

bool-literal

bool-literal ::= 'true' | 'false'

string-literal

string-literal ::= /\"[^\"]*\"/ | /\'[^\']*\'/

null-literal

null-literal ::= 'null'

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/
