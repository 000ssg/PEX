bitwise-expr

bitwise-expr ::= bitwise-or

bitwise-or

bitwise-or ::= bitwise-xor ( '|' bitwise-xor )*

bitwise-xor

bitwise-xor ::= bitwise-and ( '^' bitwise-and )*

bitwise-and

bitwise-and ::= shift-expr ( '&' shift-expr )*

shift-expr

shift-expr ::= unary-bitwise ( ( '>>>' | '<<' | '>>' ) unary-bitwise )*

unary-bitwise

unary-bitwise ::= '~' unary-bitwise | primary

primary

primary ::= int-literal | hex-literal | bin-literal | oct-literal | identifier | '(' bitwise-expr ')'

int-literal

int-literal ::= /[0-9]+/

hex-literal

hex-literal ::= /0[xX][0-9a-fA-F]+/

bin-literal

bin-literal ::= /0[bB][01]+/

oct-literal

oct-literal ::= /0[oO][0-7]+/

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/
