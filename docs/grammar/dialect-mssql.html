mssql-extension

mssql-extension ::= top-clause | variable-decl | set-variable | try-catch | output-clause | print-statement

top-clause

top-clause ::= 'SELECT' 'TOP' number 'PERCENT'? ( 'WITH' 'TIES' )? select-list

variable-decl

variable-decl ::= 'DECLARE' variable-name data-type

set-variable

set-variable ::= 'SET' variable-name '=' expression

variable-name

variable-name ::= /@[a-zA-Z_][a-zA-Z0-9_]*/

try-catch

try-catch ::= 'BEGIN' 'TRY' statement-block 'END' 'TRY' 'BEGIN' 'CATCH' statement-block 'END' 'CATCH'

output-clause

output-clause ::= 'OUTPUT' output-column-list

output-column-list

output-column-list ::= output-column ( ',' output-column )*

output-column

output-column ::= ( 'INSERTED' | 'DELETED' ) '.' column-name

print-statement

print-statement ::= 'PRINT' ( variable-name | string-literal | number )

statement-block

statement-block ::= statement+

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS'? identifier )?

data-type

data-type ::= 'INT' | 'INTEGER' | 'BIGINT' | 'FLOAT' | 'DECIMAL' | 'VARCHAR' ( '(' number ')' )? | 'NVARCHAR' ( '(' number ')' )? | 'CHAR' | 'TEXT' | 'BIT' | 'DATE' | 'DATETIME'

column-name

column-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | variable-name | identifier ( '.' identifier )? | function-call | '(' expression ')' | 'NOT' expression-term

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

statement

statement ::= expression | variable-decl | set-variable | print-statement
