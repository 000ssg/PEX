mysql-extension

mysql-extension ::= replace-into | on-duplicate-key | show-statement | describe-statement

replace-into

replace-into ::= 'REPLACE' 'INTO' table-name '(' column-list ')' 'VALUES' value-row ( ',' value-row )*

on-duplicate-key

on-duplicate-key ::= insert-statement 'ON' 'DUPLICATE' 'KEY' 'UPDATE' set-list

show-statement

show-statement ::= show-tables | show-databases | show-columns

show-tables

show-tables ::= 'SHOW' 'TABLES'

show-databases

show-databases ::= 'SHOW' 'DATABASES'

show-columns

show-columns ::= 'SHOW' 'COLUMNS' 'FROM' table-name

describe-statement

describe-statement ::= ( 'DESCRIBE' | 'DESC' ) table-name

insert-statement

insert-statement ::= 'INSERT' 'INTO' table-name '(' column-list ')' 'VALUES' value-row ( ',' value-row )*

set-list

set-list ::= set-item ( ',' set-item )*

set-item

set-item ::= column-name '=' expression

value-row

value-row ::= '(' literal-value ( ',' literal-value )* ')'

column-list

column-list ::= column-name ( ',' column-name )*

table-name

table-name ::= identifier

column-name

column-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | identifier ( '.' identifier )? | function-call | '(' expression ')' | 'NOT' expression-term

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/
