oracle-extension

oracle-extension ::= connect-by-clause | rownum-predicate | dual-query | sequence-op | create-sequence | minus-op

connect-by-clause

connect-by-clause ::= 'START' 'WITH' condition 'CONNECT' 'BY' 'NOCYCLE'? condition

rownum-predicate

rownum-predicate ::= 'ROWNUM' comparison-op number

dual-query

dual-query ::= 'SELECT' expression-list 'FROM' 'DUAL'

sequence-op

sequence-op ::= sequence-name '.' ( 'NEXTVAL' | 'CURRVAL' )

create-sequence

create-sequence ::= 'CREATE' 'SEQUENCE' sequence-name ( 'START' 'WITH' number )? ( 'INCREMENT' 'BY' number )? ( 'MINVALUE' number | 'NOMINVALUE' )? ( 'MAXVALUE' number | 'NOMAXVALUE' )? ( 'CYCLE' | 'NOCYCLE' )?

minus-op

minus-op ::= select-statement 'MINUS' select-statement

comparison-op

comparison-op ::= '=' | '<' | '>' | '<=' | '>=' | '<>' | '!='

expression-list

expression-list ::= expression ( ',' expression )*

sequence-name

sequence-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

number

number ::= /[0-9]+/

condition

condition ::= expression

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | identifier ( '.' identifier )? | function-call | '(' expression ')' | 'NOT' expression-term

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

string-literal

string-literal ::= /'[^']*'/

select-statement

select-statement ::= 'SELECT' select-list ( 'FROM' table-ref ( ',' table-ref )* )? ( 'WHERE' expression )?

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS'? identifier )?

table-ref

table-ref ::= identifier ( 'AS'? identifier )?
