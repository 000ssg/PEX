postgresql-extension

postgresql-extension ::= on-conflict-clause | returning-clause | type-cast | ilike-expr | distinct-on | generate-series | array-literal | do-block

on-conflict-clause

on-conflict-clause ::= 'ON' 'CONFLICT' '(' column-list ')' conflict-action

conflict-action

conflict-action ::= 'DO' 'NOTHING' | 'DO' 'UPDATE' 'SET' set-list

returning-clause

returning-clause ::= 'RETURNING' column-list

type-cast

type-cast ::= expression '::' type-name

ilike-expr

ilike-expr ::= expression 'NOT'? 'ILIKE' pattern

distinct-on

distinct-on ::= 'DISTINCT' 'ON' '(' column-list ')'

generate-series

generate-series ::= 'GENERATE_SERIES' '(' number ',' number ( ',' number )? ')'

array-literal

array-literal ::= 'ARRAY' '[' ( literal-value ( ',' literal-value )* )? ']'

do-block

do-block ::= 'DO' '$$' block-body '$$'

block-body

block-body ::= 'BEGIN' statement-list 'END'

set-list

set-list ::= set-item ( ',' set-item )*

set-item

set-item ::= column-name '=' expression

column-list

column-list ::= column-name ( ',' column-name )*

column-name

column-name ::= identifier

type-name

type-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

number

number ::= /[0-9]+/

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

pattern

pattern ::= string-literal

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | identifier ( '.' identifier )? | function-call | '(' expression ')' | 'NOT' expression-term

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'ILIKE' | 'IS' | '||' | '::'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

statement-list

statement-list ::= statement ( ';' statement )* ';'?

statement

statement ::= identifier ( identifier | literal-value | expression-op | '(' | ')' | ',' | '.' )*

string-literal

string-literal ::= /'[^']*'/
