math-call

math-call ::= unary-math-call | binary-math-call | constant-call

unary-math-call

unary-math-call ::= unary-math-name '(' expression ')'

unary-math-name

unary-math-name ::= 'abs' | 'ceil' | 'floor' | 'round' | 'sqrt' | 'cbrt' | 'exp' | 'log' | 'log10' | 'log2' | 'sin' | 'cos' | 'tan' | 'asin' | 'acos' | 'atan' | 'toRadians' | 'toDegrees' | 'signum'

binary-math-call

binary-math-call ::= binary-math-name '(' expression ',' expression ')'

binary-math-name

binary-math-name ::= 'pow' | 'max' | 'min' | 'atan2'

constant-call

constant-call ::= constant-name '(' ')'

constant-name

constant-name ::= 'random' | 'PI' | 'E'

expression

expression ::= additive-expr

additive-expr

additive-expr ::= multiplicative-expr ( ( '+' | '-' ) multiplicative-expr )*

multiplicative-expr

multiplicative-expr ::= unary-expr ( ( '*' | '/' | '%' ) unary-expr )*

unary-expr

unary-expr ::= ( '-' | '+' ) unary-expr | primary

primary

primary ::= number | identifier | math-call | '(' expression ')'

number

number ::= /[0-9]+(\.[0-9]+)?/

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/
