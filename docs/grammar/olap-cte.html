with-clause

with-clause ::= 'WITH' 'RECURSIVE'? cte-list select-statement

cte-list

cte-list ::= cte-definition ( ',' cte-definition )*

cte-definition

cte-definition ::= cte-name ( '(' column-alias-list ')' )? 'AS' '(' select-statement ')'

column-alias-list

column-alias-list ::= identifier ( ',' identifier )*

cte-name

cte-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

select-statement

select-statement ::= 'SELECT' select-list ( 'FROM' table-ref ( ',' table-ref )* )? ( 'WHERE' expression )? ( 'GROUP' 'BY' identifier ( ',' identifier )* )? ( 'ORDER' 'BY' identifier ( 'ASC' | 'DESC' )? ( ',' identifier ( 'ASC' | 'DESC' )? )* )? ( 'UNION' 'ALL'? select-statement )?

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS'? identifier )?

table-ref

table-ref ::= identifier ( 'AS'? identifier )?

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= identifier ( '.' identifier )? | number | string-literal | function-call | '(' expression ')'

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/
