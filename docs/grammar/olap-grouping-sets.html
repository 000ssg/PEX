grouping-extension

grouping-extension ::= grouping-sets | cube-clause | rollup-clause

grouping-sets

grouping-sets ::= 'GROUPING' 'SETS' '(' grouping-element-list ')'

grouping-element-list

grouping-element-list ::= grouping-element ( ',' grouping-element )*

grouping-element

grouping-element ::= '(' column-list? ')' | column-name

cube-clause

cube-clause ::= 'CUBE' '(' column-list ')'

rollup-clause

rollup-clause ::= 'ROLLUP' '(' column-list ')'

column-list

column-list ::= column-name ( ',' column-name )*

column-name

column-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/
