merge-statement

merge-statement ::= 'MERGE' 'INTO' target-table ( 'AS'? alias )? 'USING' source-table ( 'AS'? alias )? 'ON' condition merge-action+

merge-action

merge-action ::= when-matched | when-not-matched

when-matched

when-matched ::= 'WHEN' 'MATCHED' 'THEN' ( merge-update | merge-delete )

when-not-matched

when-not-matched ::= 'WHEN' 'NOT' 'MATCHED' 'THEN' merge-insert

merge-update

merge-update ::= 'UPDATE' 'SET' set-list

merge-delete

merge-delete ::= 'DELETE'

merge-insert

merge-insert ::= 'INSERT' ( '(' column-list ')' )? 'VALUES' '(' value-list ')'

set-list

set-list ::= set-item ( ',' set-item )*

set-item

set-item ::= ( table-ref '.' )? column-name '=' expression

column-list

column-list ::= column-name ( ',' column-name )*

value-list

value-list ::= expression ( ',' expression )*

target-table

target-table ::= identifier

source-table

source-table ::= identifier

table-ref

table-ref ::= identifier

column-name

column-name ::= identifier

alias

alias ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

condition

condition ::= expression

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | identifier ( '.' identifier )? | function-call | '(' expression ')' | 'NOT' expression-term

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/
