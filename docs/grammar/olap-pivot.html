pivot-clause

pivot-clause ::= 'PIVOT' '(' aggregate-func '(' column-name ')' 'FOR' column-name 'IN' '(' value-list ')' ')'

unpivot-clause

unpivot-clause ::= 'UNPIVOT' '(' value-column 'FOR' name-column 'IN' '(' column-list ')' ')'

aggregate-func

aggregate-func ::= 'SUM' | 'AVG' | 'COUNT' | 'MIN' | 'MAX'

value-list

value-list ::= literal-value ( ',' literal-value )*

column-list

column-list ::= column-name ( ',' column-name )*

value-column

value-column ::= identifier

name-column

name-column ::= identifier

column-name

column-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

literal-value

literal-value ::= number | string-literal | 'NULL'

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/
