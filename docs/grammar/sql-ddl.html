ddl-statement

ddl-statement ::= create-statement | alter-statement | drop-statement

create-statement

create-statement ::= create-table | create-index | create-view | create-trigger | create-procedure

create-table

create-table ::= 'CREATE' 'TABLE' ( 'IF' 'NOT' 'EXISTS' )? table-name '(' table-element-list ')'

table-element-list

table-element-list ::= table-element ( ',' table-element )*

table-element

table-element ::= column-def | table-constraint

column-def

column-def ::= column-name data-type column-constraint*

column-constraint

column-constraint ::= 'NOT' 'NULL' | 'NULL' | 'DEFAULT' literal-value | 'AUTO_INCREMENT' | 'SERIAL' | 'PRIMARY' 'KEY' | 'UNIQUE' | 'REFERENCES' table-name ( '(' column-list ')' )?

table-constraint

table-constraint ::= ( 'CONSTRAINT' constraint-name )? ( primary-key-constraint | foreign-key-constraint | unique-constraint | check-constraint )

primary-key-constraint

primary-key-constraint ::= 'PRIMARY' 'KEY' '(' column-list ')'

foreign-key-constraint

foreign-key-constraint ::= 'FOREIGN' 'KEY' '(' column-list ')' 'REFERENCES' table-name '(' column-list ')' ( 'ON' ( 'DELETE' | 'UPDATE' ) referential-action )*

referential-action

referential-action ::= 'CASCADE' | 'SET' 'NULL' | 'RESTRICT' | 'NO' 'ACTION'

unique-constraint

unique-constraint ::= 'UNIQUE' '(' column-list ')'

check-constraint

check-constraint ::= 'CHECK' '(' expression ')'

create-index

create-index ::= 'CREATE' 'UNIQUE'? 'INDEX' index-name 'ON' table-name '(' column-list ')'

create-view

create-view ::= 'CREATE' ( 'OR' 'REPLACE' )? 'VIEW' view-name 'AS' select-statement

create-trigger

create-trigger ::= 'CREATE' 'TRIGGER' trigger-name trigger-timing trigger-event 'ON' table-name ( 'BEGIN' statement-list 'END' )?

trigger-timing

trigger-timing ::= 'BEFORE' | 'AFTER'

trigger-event

trigger-event ::= 'INSERT' | 'UPDATE' | 'DELETE'

create-procedure

create-procedure ::= 'CREATE' ( 'OR' 'REPLACE' )? 'PROCEDURE' proc-name '(' param-list? ')' ( 'BEGIN' statement-list 'END' )?

param-list

param-list ::= procedure-param ( ',' procedure-param )*

procedure-param

procedure-param ::= param-mode? param-name data-type

param-mode

param-mode ::= 'IN' | 'OUT' | 'INOUT' | 'IN' 'OUT'

alter-statement

alter-statement ::= 'ALTER' 'TABLE' table-name alter-action-list

alter-action-list

alter-action-list ::= alter-action ( ',' alter-action )*

alter-action

alter-action ::= add-column | drop-column | rename-column | add-constraint

add-column

add-column ::= 'ADD' 'COLUMN'? column-def

drop-column

drop-column ::= 'DROP' 'COLUMN'? column-name

rename-column

rename-column ::= 'RENAME' 'COLUMN'? column-name 'TO' column-name

add-constraint

add-constraint ::= 'ADD' table-constraint

drop-statement

drop-statement ::= 'DROP' 'TABLE' ( 'IF' 'EXISTS' )? table-name

data-type

data-type ::= 'INT' | 'INTEGER' | 'BIGINT' | 'FLOAT' | 'REAL' | 'DOUBLE' | 'DECIMAL' ( '(' number ( ',' number )? ')' )? | 'NUMERIC' | 'VARCHAR' ( '(' number ')' )? | 'CHAR' | 'TEXT' | 'BOOLEAN' | 'BOOL' | 'DATE' | 'TIMESTAMP' | 'DATETIME' | 'BLOB' | 'BYTEA' | 'SERIAL'

column-list

column-list ::= column-name ( ',' column-name )*

statement-list

statement-list ::= statement+

table-name

table-name ::= identifier

column-name

column-name ::= identifier

index-name

index-name ::= identifier

view-name

view-name ::= identifier

trigger-name

trigger-name ::= identifier

proc-name

proc-name ::= identifier

constraint-name

constraint-name ::= identifier

param-name

param-name ::= identifier

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | column-name | function-call | '(' expression ')' | 'NOT' expression-term

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | 'IN'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

statement

statement ::= ddl-statement | dml-keyword identifier ( identifier | literal-value | '(' | ')' | ',' | '.' )*

dml-keyword

dml-keyword ::= 'SELECT' | 'INSERT' | 'UPDATE' | 'DELETE' | 'SET'

select-statement

select-statement ::= 'SELECT' select-list ( 'FROM' identifier ( ',' identifier )* )? ( 'WHERE' expression )?

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS' identifier )?
