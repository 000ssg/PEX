dml-statement

dml-statement ::= select-statement | insert-statement | update-statement | delete-statement

select-statement

select-statement ::= 'SELECT' 'DISTINCT'? select-list ( 'FROM' from-clause )? ( 'WHERE' where-clause )? ( 'GROUP' 'BY' group-by-list )? ( 'HAVING' having-clause )? ( 'ORDER' 'BY' order-by-list )? ( 'LIMIT' limit-clause )?

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS'? alias )?

from-clause

from-clause ::= table-ref ( ',' table-ref )*

table-ref

table-ref ::= table-name ( 'AS'? alias )? join-clause?

join-clause

join-clause ::= join-type 'JOIN' table-name ( 'AS'? alias )? ( 'ON' expression )?

join-type

join-type ::= 'INNER' | 'LEFT' 'OUTER'? | 'RIGHT' 'OUTER'? | 'FULL' 'OUTER'? | 'CROSS'

where-clause

where-clause ::= expression

group-by-list

group-by-list ::= column-ref ( ',' column-ref )*

having-clause

having-clause ::= expression

order-by-list

order-by-list ::= order-by-item ( ',' order-by-item )*

order-by-item

order-by-item ::= column-ref ( 'ASC' | 'DESC' )? ( 'NULLS' ( 'FIRST' | 'LAST' ) )?

limit-clause

limit-clause ::= number ( 'OFFSET' number )? | number ',' number

insert-statement

insert-statement ::= 'INSERT' 'INTO' table-name ( '(' column-list ')' )? ( values-clause | select-statement )

values-clause

values-clause ::= 'VALUES' value-row ( ',' value-row )*

value-row

value-row ::= '(' literal-value ( ',' literal-value )* ')'

update-statement

update-statement ::= 'UPDATE' table-name 'SET' set-clause-list ( 'WHERE' where-clause )?

set-clause-list

set-clause-list ::= set-clause ( ',' set-clause )*

set-clause

set-clause ::= column-name '=' literal-value

delete-statement

delete-statement ::= 'DELETE' 'FROM' table-name ( 'WHERE' where-clause )?

call-statement

call-statement ::= 'CALL' proc-name '(' ( literal-value ( ',' literal-value )* )? ')'

column-list

column-list ::= column-name ( ',' column-name )*

column-ref

column-ref ::= ( table-name '.' )? column-name

table-name

table-name ::= identifier

column-name

column-name ::= identifier

proc-name

proc-name ::= identifier

alias

alias ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | column-ref | function-call | '(' expression ')' | 'NOT' expression-term | '*'

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '%' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | 'IN' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'
