expression

expression ::= or-expression

or-expression

or-expression ::= and-expression ( 'OR' and-expression )*

and-expression

and-expression ::= not-expression ( 'AND' not-expression )*

not-expression

not-expression ::= 'NOT' not-expression | comparison

comparison

comparison ::= addition comparison-tail?

comparison-tail

comparison-tail ::= comparison-op addition | 'IS' 'NOT'? 'NULL' | 'NOT'? 'IN' '(' in-list ')' | 'NOT'? 'BETWEEN' addition 'AND' addition | 'NOT'? 'LIKE' string-literal | 'NOT'? 'ILIKE' string-literal

comparison-op

comparison-op ::= '=' | '<' | '>' | '<=' | '>=' | '<>' | '!='

in-list

in-list ::= expression ( ',' expression )* | select-statement

addition

addition ::= multiplication ( ( '+' | '-' | '||' ) multiplication )*

multiplication

multiplication ::= unary ( ( '*' | '/' | '%' ) unary )*

unary

unary ::= ( '-' | '+' ) primary | primary

primary

primary ::= literal | aggregate-call | function-call | case-expression | exists-expression | subquery | column-ref | '(' expression ')'

aggregate-call

aggregate-call ::= aggregate-name '(' 'DISTINCT'? ( expression | '*' ) ')'

aggregate-name

aggregate-name ::= 'COUNT' | 'SUM' | 'AVG' | 'MIN' | 'MAX' | 'GROUP_CONCAT'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* )? ')'

case-expression

case-expression ::= 'CASE' expression? when-clause+ ( 'ELSE' expression )? 'END'

when-clause

when-clause ::= 'WHEN' expression 'THEN' expression

exists-expression

exists-expression ::= 'EXISTS' '(' select-statement ')'

subquery

subquery ::= '(' select-statement ')'

column-ref

column-ref ::= ( table-name '.' )? column-name

literal

literal ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'

number

number ::= /[0-9]+(\.[0-9]+)?/

string-literal

string-literal ::= /'[^']*'/

table-name

table-name ::= identifier

column-name

column-name ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

select-statement

select-statement ::= 'SELECT' select-list ( 'FROM' table-ref-list )? ( 'WHERE' expression )? ( 'GROUP' 'BY' column-ref ( ',' column-ref )* )? ( 'ORDER' 'BY' column-ref ( 'ASC' | 'DESC' )? ( ',' column-ref ( 'ASC' | 'DESC' )? )* )?

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS'? identifier )?

table-ref-list

table-ref-list ::= table-ref ( ',' table-ref )*

table-ref

table-ref ::= table-name ( 'AS'? identifier )?
