streaming-statement

streaming-statement ::= create-stream | drop-stream | stream-select | insert-into-stream

create-stream

create-stream ::= 'CREATE' 'STREAM' stream-name '(' column-def-list ')' ( 'WITH' '(' property-list ')' )?

column-def-list

column-def-list ::= column-def ( ',' column-def )*

column-def

column-def ::= column-name data-type

property-list

property-list ::= property ( ',' property )*

property

property ::= property-name '=' property-value

property-name

property-name ::= identifier

property-value

property-value ::= string-literal | number | identifier

drop-stream

drop-stream ::= 'DROP' 'STREAM' ( 'IF' 'EXISTS' )? stream-name

stream-select

stream-select ::= 'SELECT' select-list 'FROM' stream-name ( 'WINDOW' window-spec )? ( 'WHERE' condition )? ( 'GROUP' 'BY' column-list )? ( 'EMIT' emit-strategy )?

select-list

select-list ::= select-item ( ',' select-item )*

select-item

select-item ::= '*' | expression ( 'AS' alias )?

window-spec

window-spec ::= tumbling-window | hopping-window | sliding-window | session-window

tumbling-window

tumbling-window ::= 'TUMBLING' '(' 'SIZE' duration ')'

hopping-window

hopping-window ::= 'HOPPING' '(' 'SIZE' duration ',' 'ADVANCE' duration ')'

sliding-window

sliding-window ::= 'SLIDING' '(' 'SIZE' duration ')'

session-window

session-window ::= 'SESSION' '(' 'GAP' duration ')'

duration

duration ::= number time-unit

time-unit

time-unit ::= 'MILLISECONDS' | 'MILLIS' | 'MS' | 'SECONDS' | 'SECOND' | 'SEC' | 'MINUTES' | 'MINUTE' | 'MIN' | 'HOURS' | 'HOUR' | 'DAYS' | 'DAY'

emit-strategy

emit-strategy ::= 'CHANGES' | 'FINAL'

insert-into-stream

insert-into-stream ::= 'INSERT' 'INTO' stream-name stream-select

data-type

data-type ::= 'INT' | 'INTEGER' | 'BIGINT' | 'FLOAT' | 'REAL' | 'DOUBLE' | 'DECIMAL' | 'NUMERIC' | 'VARCHAR' ( '(' number ')' )? | 'CHAR' | 'TEXT' | 'BOOLEAN' | 'BOOL' | 'DATE' | 'TIMESTAMP' | 'DATETIME'

column-list

column-list ::= column-name ( ',' column-name )*

stream-name

stream-name ::= identifier

column-name

column-name ::= identifier

alias

alias ::= identifier

identifier

identifier ::= /[a-zA-Z_][a-zA-Z0-9_]*/

number

number ::= /[0-9]+/

string-literal

string-literal ::= /'[^']*'/

condition

condition ::= expression

expression

expression ::= expression-term ( expression-op expression-term )*

expression-term

expression-term ::= literal-value | column-name | function-call | '(' expression ')' | 'NOT' expression-term | '*'

expression-op

expression-op ::= '+' | '-' | '*' | '/' | '=' | '<' | '>' | '<=' | '>=' | '<>' | '!=' | 'AND' | 'OR' | 'LIKE' | 'IS' | '||'

function-call

function-call ::= identifier '(' ( expression ( ',' expression )* | '*' )? ')'

literal-value

literal-value ::= number | string-literal | 'NULL' | 'TRUE' | 'FALSE'
